    Atividade 4 — Autoencoders Variacionais (VAE)

    Exercício 1

    A — Carregue e divida os dados

    Utilizei torchvision.datasets.MNIST, transforms.ToTensor() e vetores de 784 pixels em [0,1]. Separei os 60.000 exemplos oficiais de treino em 50.000 de treino e 10.000 de validação, usando random_split(..., generator=torch.Generator().manual_seed(42)). Reservei as 10.000 imagens oficiais de teste para a avaliação final e as visualizações solicitadas.

    from pathlib import Path
import torch
from torch.utils.data import DataLoader, random_split
from torchvision import datasets, transforms

ROOT = Path(__file__).resolve().parent.parent
BATCH = 128

def carregar_dados():
    transform = transforms.ToTensor()
    full = datasets.MNIST(str(ROOT / 'data'), train=True, download=True, transform=transform)
    test = datasets.MNIST(str(ROOT / 'data'), train=False, download=True, transform=transform)
    train, val = random_split(full, [50_000, 10_000],
                              generator=torch.Generator().manual_seed(42))
    train_loader = DataLoader(train, batch_size=BATCH, shuffle=True)
    val_loader = DataLoader(val, batch_size=BATCH, shuffle=False)
    test_loader = DataLoader(test, batch_size=BATCH, shuffle=False)
    return train_loader, val_loader, test_loader, test

    B — Construa o modelo

    O encoder é 784 → 512 → 256, com ReLU, e produz duas cabeças 256 → d: média μ e log-variância log σ². O decoder é d → 256 → 512 → 784, com ReLU nas camadas ocultas e sigmoid na saída. Aqui, d=16, β=1.

    A reparametrização é z = μ + exp(0,5 log σ²) ε, onde ε ~ N(0,I). Calculo separadamente a BCE somada nos pixels e a KL fechada:

    \[ \mathrm{BCE}=-\frac{1}{N}\sum_{n=1}^{N}\sum_{j=1}^{784}\left[x_{nj}\log \hat x_{nj}+(1-x_{nj})\log(1-\hat x_{nj})\right] \]
    \[ \mathrm{KL}=-\frac{1}{2N}\sum_{n=1}^{N}\sum_{i=1}^{d}(1+\log \sigma_{ni}^2-\mu_{ni}^2-\sigma_{ni}^2),\quad \mathcal{L}=\mathrm{BCE}+\mathrm{KL}. \]
    import torch
from torch import nn
from torch.nn import functional as F

class VAE(nn.Module):
    def __init__(self, d):
        super().__init__()
        self.encoder = nn.Sequential(nn.Linear(784, 512), nn.ReLU(), nn.Linear(512, 256), nn.ReLU())
        self.mu = nn.Linear(256, d)
        self.logvar = nn.Linear(256, d)
        self.decoder = nn.Sequential(nn.Linear(d, 256), nn.ReLU(), nn.Linear(256, 512), nn.ReLU(), nn.Linear(512, 784), nn.Sigmoid())

    def encode(self, x):
        h = self.encoder(x)
        return self.mu(h), self.logvar(h)

    def reparameterize(self, mu, logvar):
        eps = torch.randn_like(mu)
        return mu + torch.exp(0.5 * logvar) * eps

    def forward(self, x):
        mu, logvar = self.encode(x)
        z = self.reparameterize(mu, logvar)
        return self.decoder(z), mu, logvar

def loss_terms(x, xhat, mu, logvar):
    reconstruction = F.binary_cross_entropy(xhat, x, reduction='sum') / x.size(0)
    kl = -0.5 * torch.sum(1 + logvar - mu.pow(2) - logvar.exp()) / x.size(0)
    return reconstruction, kl

    C — Treine

    Treinei com Adam, taxa de aprendizado 0,001, batches de 128 e 20 épocas. Registrei reconstrução e KL em treino e validação, separadamente. No teste oficial, obtive BCE = 79.981, KL = 20.691 e ELBO negativo = 100.672 nats/imagem.

    from pathlib import Path
import numpy as np
import torch
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from ex1b import VAE, loss_terms

ROOT = Path(__file__).resolve().parent.parent
FIG = ROOT / 'figures'
FIG.mkdir(exist_ok=True)
EPOCHS = 20
LR = 1e-3
DEVICE = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

def run_epoch(model, loader, optimizer=None):
    train = optimizer is not None
    model.train(train)
    totals = np.zeros(2, dtype=np.float64)
    count = 0
    with torch.set_grad_enabled(train):
        for x, _ in loader:
            x = x.to(DEVICE).flatten(1)
            xhat, mu, logvar = model(x)
            reconstruction, kl = loss_terms(x, xhat, mu, logvar)
            if train:
                optimizer.zero_grad()
                (reconstruction + kl).backward()
                optimizer.step()
            n = x.size(0)
            totals += n * np.array([reconstruction.item(), kl.item()])
            count += n
    return (totals / count).tolist()

def train_model(d, train_loader, val_loader):
    torch.manual_seed(42)  
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(42)
    model = VAE(d).to(DEVICE)
    optimizer = torch.optim.Adam(model.parameters(), lr=LR)
    hist = {'train_rec': [], 'train_kl': [], 'val_rec': [], 'val_kl': []}
    for epoch in range(EPOCHS):
        a, b = run_epoch(model, train_loader, optimizer)
        c, e = run_epoch(model, val_loader)
        for key, value in zip(hist.keys(), (a, b, c, e)):
            hist[key].append(value)
        print(f'd={d:2d} epoca {epoch+1:02d}/{EPOCHS}: treino rec={a:.3f} KL={b:.3f}; val rec={c:.3f} KL={e:.3f}', flush=True)
    return model, hist

def fig1(hist):
    fig, axs = plt.subplots(1, 2, figsize=(11, 4))
    for ax, metric, title in [(axs[0], 'rec', 'Reconstrucao (BCE)'), (axs[1], 'kl', 'Divergencia KL')]:
        ax.plot(range(1, EPOCHS+1), hist['train_'+metric], label='Treino')
        ax.plot(range(1, EPOCHS+1), hist['val_'+metric], label='Validacao')
        ax.set(xlabel='Epoca', ylabel='nats por imagem', title=title)
        ax.legend()
    fig.suptitle('Figura 1 — Perdas do VAE d=16')
    fig.tight_layout()
    fig.savefig(FIG/'fig1.png', dpi=160)
    plt.close(fig)

    Figura 1 — Evolução da reconstrução e da KL para o VAE de dimensão latente 16.

    D — Reconstruções e amostras

    Na Figura 2, cada coluna traz a primeira imagem de teste de um dígito (0 a 9) e sua reconstrução obtida ao decodificar μ, sem sortear z.

    from pathlib import Path
import torch
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from ex1c import DEVICE, FIG

def display_img(ax, flat):
    ax.imshow(flat.reshape(28, 28), cmap='gray', vmin=0, vmax=1)
    ax.axis('off')

def fig2(model, test):
    indices = {}
    for i, (_, y) in enumerate(test):
        digit = int(y)
        if digit not in indices:
            indices[digit] = i
        if len(indices) == 10:
            break
    images = torch.stack([test[indices[d]][0] for d in range(10)]).to(DEVICE).flatten(1)
    with torch.no_grad():
        mu, _ = model.encode(images)
        outputs = model.decoder(mu)
    fig, ax = plt.subplots(2, 10, figsize=(13, 3.3))
    for d in range(10):
        display_img(ax[0, d], images[d].cpu().numpy())
        display_img(ax[1, d], outputs[d].cpu().numpy())
        ax[0, d].set_title(str(d))
    ax[0, 0].set_ylabel('Original', fontsize=10)
    ax[1, 0].set_ylabel('Reconstrucao', fontsize=10)
    fig.suptitle('Figura 2 — Originais e reconstrucoes a partir de mu (d=16)')
    fig.tight_layout()
    fig.savefig(FIG/'fig2.png', dpi=160)
    plt.close(fig)
    return indices

def fig3(model):
    torch.manual_seed(0)
    with torch.no_grad():
        z = torch.randn(64, 16, device=DEVICE)
        samples = model.decoder(z).cpu().numpy()
    fig, axs = plt.subplots(8, 8, figsize=(8, 8))
    for ax, sample in zip(axs.ravel(), samples):
        display_img(ax, sample)
    fig.suptitle('Figura 3 — 64 amostras do prior N(0,I), d=16')
    fig.tight_layout()
    fig.savefig(FIG/'fig3.png', dpi=160)
    plt.close(fig)

    E — Análise

    1. Por que a KL pode subir enquanto a perda total cai? O treinamento minimiza a soma BCE + KL. O encoder pode tornar os códigos mais informativos sobre cada imagem, reduzindo bastante a BCE, mesmo que isso afaste a posteriori aproximada do prior e aumente a KL. A troca compensa quando a queda na reconstrução é maior que o aumento na regularização. A KL de validação passou de 13.343 na primeira época para 20.674 nats/imagem na última.

    2. Por que as amostras da Figura 3 podem parecer piores que as reconstruções da Figura 2? Primeiro, nem toda região do prior necessariamente corresponde a códigos onde o encoder colocou exemplos de treino: amostras aleatórias podem cair em regiões pouco representadas e gerar imagens ambíguas. Segundo, o decoder prevê probabilidades de pixels; ao representar alternativas possíveis, seus valores intermediários podem produzir imagens borradas, sobretudo quando o código não identifica claramente um dígito.

    3. Por que decodificar μ na Figura 2? Para reconstruir de modo determinístico a representação central de cada entrada. Amostrar z acrescentaria ruído e dificultaria a comparação entre a imagem de entrada e sua reconstrução.

    Exercício 2

    A — Treine um VAE 2-D

    Repeti a arquitetura, a divisão, β=1, a semente 42 e as 20 épocas; a única alteração foi d=2. Os resultados finais no teste são:

    Dimensão latente Reconstrução (nats/imagem) KL (nats/imagem)
    16 79.981 20.691
    2 134.121 6.746 
    from ex1c import train_model, run_epoch

def treinar_vae_2d(train_loader, val_loader, test_loader):
    model, history = train_model(2, train_loader, val_loader)
    reconstruction, kl = run_epoch(model, test_loader)
    return model, history, reconstruction, kl

    B — O conjunto de teste codificado

    A Figura 4 apresenta as médias μ das 10.000 imagens de teste, usando cores e legenda para identificar cada dígito.

    import numpy as np
import torch
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from ex1c import DEVICE, FIG

def fig4(model, test_loader):
    points, digits = [], []
    with torch.no_grad():
        for x, y in test_loader:
            mu, _ = model.encode(x.to(DEVICE).flatten(1))
            points.append(mu.cpu().numpy())
            digits.append(y.numpy())
    points = np.concatenate(points)
    digits = np.concatenate(digits)
    fig, ax = plt.subplots(figsize=(9, 7))
    for d in range(10):
        select = digits == d
        ax.scatter(points[select, 0], points[select, 1], s=3, alpha=0.45, label=str(d))
    ax.set(xlabel='mu_1', ylabel='mu_2', title='Figura 4 — Medias latentes das 10.000 imagens de teste (d=2)')
    ax.legend(title='Digito', markerscale=4, ncol=2)
    fig.tight_layout()
    fig.savefig(FIG/'fig4.png', dpi=160)
    plt.close(fig)

    C — O plano decodificado

    Na Figura 5, decodifiquei uma grade 15 × 15 com coordenadas norm.ppf(q), onde q assume 15 valores igualmente espaçados entre 0,05 e 0,95.

    import numpy as np
import torch
from scipy.stats import norm
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from ex1c import DEVICE, FIG

def fig5(model):
    coordinates = norm.ppf(np.linspace(0.05, 0.95, 15))
    xx, yy = np.meshgrid(coordinates, coordinates[::-1])
    zs = torch.tensor(np.stack([xx.ravel(), yy.ravel()], axis=1), dtype=torch.float32, device=DEVICE)
    with torch.no_grad():
        decoded = model.decoder(zs).cpu().numpy().reshape(15, 15, 28, 28)
    canvas = np.zeros((15*28, 15*28))
    for i in range(15):
        for j in range(15):
            canvas[i*28:(i+1)*28, j*28:(j+1)*28] = decoded[i, j]
    fig, ax = plt.subplots(figsize=(10, 10))
    ax.imshow(canvas, cmap='gray', extent=[coordinates[0], coordinates[-1], coordinates[0], coordinates[-1]], origin='upper', vmin=0, vmax=1)
    ax.set(xlabel='z_1 (quantis N(0,1))', ylabel='z_2 (quantis N(0,1))', title='Figura 5 — Grade de decodificacao 15 x 15 (d=2)')
    fig.tight_layout()
    fig.savefig(FIG/'fig5.png', dpi=160)
    plt.close(fig)

    D — Interpolação

    Escolhi os índices de teste 0 (dígito 7) e 2 (dígito 1), que correspondem a dígitos diferentes. A Figura 6 compara dez passos lineares em pixels com dez passos entre μ_a e μ_b, decodificados pelo VAE de 16 dimensões.

    import torch
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
from ex1c import DEVICE, FIG
from ex1d import display_img

def fig6(model, test):
    ia, ib = 0, 2
    xa, ya = test[ia]
    xb, yb = test[ib]
    assert int(ya) != int(yb)
    xa, xb = xa.to(DEVICE).flatten(), xb.to(DEVICE).flatten()
    ts = torch.linspace(0, 1, 10, device=DEVICE).unsqueeze(1)
    with torch.no_grad():
        mua, _ = model.encode(xa.unsqueeze(0))
        mub, _ = model.encode(xb.unsqueeze(0))
        pixel_blend = ((1-ts)*xa + ts*xb).cpu().numpy()
        latent_blend = model.decoder((1-ts)*mua + ts*mub).cpu().numpy()
    fig, axs = plt.subplots(2, 10, figsize=(14, 3.5))
    for j in range(10):
        display_img(axs[0, j], pixel_blend[j])
        display_img(axs[1, j], latent_blend[j])
        axs[0, j].set_title(f't={j/9:.2f}', fontsize=9)
    axs[0, 0].set_ylabel('Pixels')
    axs[1, 0].set_ylabel('Latente')
    fig.suptitle(f'Figura 6 — Interpolacao dos indices {ia} (digito {int(ya)}) e {ib} (digito {int(yb)})')
    fig.tight_layout()
    fig.savefig(FIG/'fig6.png', dpi=160)
    plt.close(fig)
    return [ia, ib], [int(ya), int(yb)]

    Figura 6 — Mistura direta de pixels (linha superior) e interpolação no espaço latente de dimensão 16 (linha inferior).

    E — Análise

    1. Sobreposição de dígitos e transição no plano. Na Figura 4, há sobreposição entre diferentes classes, por exemplo entre os dígitos 4 e 9 na região central do espaço latente, aproximadamente para valores de μ_1 e μ_2 entre 0 e 1. Na Figura 5, na região central-direita, observam-se formas que passam gradualmente de traços semelhantes ao 4 para traços semelhantes ao 9. Algumas imagens intermediárias misturam características dos dois dígitos, dificultando a identificação de uma única classe.

    2. Por que d=2 reconstrói pior? Com apenas duas dimensões latentes, o modelo precisa comprimir informações sobre a identidade dos dígitos e sobre o estilo de escrita, como curvatura, inclinação, espessura e posição dos traços. Parte dessas características é perdida ou não pode ser representada separadamente. Na Figura 5, as regiões de transição apresentam imagens com traços misturados ou borrados, evidenciando essa limitação. A reconstrução no teste foi de 134.121 nats/imagem para d=2, contra 79.981 nats/imagem para d=16, uma diferença de 54.139 nats/imagem.

    3. Comparação entre as interpolações. Na Figura 6, a interpolação em pixels (linha superior) faz uma sobreposição direta das imagens dos dígitos 7 e 1. Nas etapas intermediárias, aparecem traços de ambos ao mesmo tempo, formando imagens ambíguas. Na interpolação latente (linha inferior), a transformação é mais coerente: o 7 perde gradualmente o traço superior e assume a forma do 1. Isso ocorre porque o decoder transforma representações intermediárias aprendidas em imagens com características de dígitos, enquanto a interpolação em pixels apenas mistura intensidades nas mesmas posições.
